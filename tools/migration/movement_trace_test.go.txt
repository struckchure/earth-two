package character

import (
    "encoding/json"
    "os"
    "path/filepath"
    "testing"
    rl "github.com/gen2brain/raylib-go/raylib"
)

// Diagnostic only: the existing reference harness and controller are unchanged.
func TestMigrationMovementTrace(t *testing.T) {
    input, out := os.Getenv("EARTH_TWO_MOVEMENT_INPUT"), os.Getenv("EARTH_TWO_MOVEMENT_OUT")
    if input == "" || out == "" { t.Fatal("movement trace paths are required") }
    var config struct {
        Cases []struct {
            Name string
            Phases []struct {
                Ticks int
                Move [3]float32
                Run, Crouch, Jump, Roll, Slide bool
            }
        }
    }
    bytes, err := os.ReadFile(input); if err != nil { t.Fatal(err) }
    if err := json.Unmarshal(bytes, &config); err != nil { t.Fatal(err) }
    for _, c := range config.Cases { t.Run(c.Name, func(t *testing.T) {
        h := newTraversalHarness(t, rl.Vector3{}, nil)
        samples := []map[string]any{}
        for _, phase := range c.Phases {
            for tick := 0; tick < phase.Ticks; tick++ {
                *h.in = Intent{Move:rl.Vector3{X:phase.Move[0], Y:phase.Move[1], Z:phase.Move[2]}, Run:phase.Run, Crouch:phase.Crouch, Jump:phase.Jump && tick==0, Roll:phase.Roll && tick==0, Slide:phase.Slide && tick==0}
                h.tick(1)
                p,v:=h.tr.Translation,h.cc.Velocity
                samples=append(samples,map[string]any{"tick":len(samples)+1,"position":[3]float32{p.X,p.Y,p.Z},"velocity":[3]float32{v.X,v.Y,v.Z},"height":h.cc.Height,"grounded":h.cc.Grounded,"mode":uint8(h.s.Mode)})
            }
        }
        bytes,err:=json.MarshalIndent(samples,"","  ");if err!=nil{t.Fatal(err)}
        if err=os.WriteFile(filepath.Join(out,c.Name+".json"),bytes,0644);err!=nil{t.Fatal(err)}
    }) }
}
